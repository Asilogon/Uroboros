from deepseek_api import DeepSeekClient

# Инициализируем клиент (сессия автоматически подгружается из data/session.json)
client = DeepSeekClient()

# 1. Создаем чат
chat_id = client.create_chat()

# 2. Отправляем сообщение со стримингом
for chunk in client.send_message(chat_id, "Привет! Расскажи о себе.", stream=True):
    print(chunk["content"], end="", flush=True)

# 3. Прикрепляем файл
file_info = client.upload_file("document.pdf")
for chunk in client.send_message(chat_id, "Сделай выжимку документа", ref_file_ids=[file_info["id"]]):
    print(chunk["content"], end="", flush=True)

# 4. Удаляем чат
client.delete_chat(chat_id)
